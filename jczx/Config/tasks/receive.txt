/ 杂项
/ ===== method：若基础图周围出现邻居图则点击（通用：匹配基础图后按 expand 外扩边级联查 neighbor）=====
[if-around-click]
type: method
name: 若基础图周围出现邻居图则点击
params: base,base_per,neighbor,then_entity,expand_up,expand_down,expand_left,expand_right,click_sleep
param_defaults: neighbor=locations\hasNew.png,base_per=0.8,then_entity=click-target,expand_up=40,expand_down=40,expand_left=40,expand_right=40,click_sleep=1
action: matched-around,cond-click

[matched-around]
type: match
match: match-around
target: %{neighbor}
context_key: near_ok
context_type: bool

[match-around]
type: match
target: %{base}
per: %{base_per}
action: up-M|%{expand_up},down-M|%{expand_down},left-M|%{expand_left},right-M|%{expand_right}

[cond-click]
type: condition
condition: &{%{near_ok}}
condition_then: %{then_entity}

[click-target]
type: click
target: %{base}
per: %{base_per}
sleep: %{click_sleep}

[click-exploration]
type: click
name: 勘探指南
target: buttons\ExplorationGuidelines.png
wait_target: buttons\EGTask.png
max_wait: 4
