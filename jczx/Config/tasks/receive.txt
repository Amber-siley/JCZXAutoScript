/ 领取各种东西

[task-receive-everyday]
type: task
view: on
name: 领取每日礼包
action: goto-home,goto-supplyStation,click-giftPackage,wait-5,condition-to-get-freePackage

[task-receive-mail]
type: task
view: on
name: 领取邮件礼包
action: goto-home,click-mail,click-claimAll,condition-to-get-item,goto-home

[task-receive-dayAndWeek]
type: task
view: on
name: 领取常规任务奖励
action: goto-home,click-task-menu,condition-to-receive-daily,wait-1,condition-to-receive-week,goto-home

/ 杂项
[click-task-menu]
type: click
name: 任务
target: buttons\Task.png
sleep: 2

[condition-to-receive-daily]
type: condition
name: 条件判断是否有日常领取
condition: matched-daily-has-new
condition_then: click-Daily,click-claimAll,condition-to-get-item

[condition-to-receive-week]
type: condition
name: 条件判断是否有周常领取
condition: matched-week-has-new
condition_then: click-Week,click-claimAll,condition-to-get-item

[click-Daily]
type: click
target: buttons\Daily.png
sleep: 1

[click-Week]
type: click
target: buttons\Week.png
sleep: 1

[matched-daily-has-new]
type: match
match: match-Daily
target: locations\hasNew.png

[matched-week-has-new]
type: match
match: match-Week
target: locations\hasNew.png

[match-Week]
type: match
target: buttons\Week.png
action: right|1,right-M|20,up-M|10,down-M|10

[match-Daily]
type: match
target: buttons\Daily.png
action: right|1,right-M|20,up-M|10,down-M|10

[click-mail]
type: click
name: 邮件
target: buttons\mail.png
sleep: 1

[click-claimAll]
type: click
name: 一键领取
target: buttons\claimAll.png
per: 0.7
sleep: 1

[condition-to-get-item]
type: condition
name: 条件判断有领取物品
condition: in_location-getItems1
condition_then: click-get-item

[in_location-getItems1]
extend: in_location
target: buttons\getItems1.png

[goto-supplyStation]
type: click
name: 补给站
target: buttons\supplyStation.png
wait_target: buttons\giftPackage.png

[click-giftPackage]
type: click
name: 礼包
target: buttons\giftPackage.png
sleep: 1

[condition-to-get-freePackage]
type: condition
name: 条件判断是否有免费礼包
condition: condition-in_location-freePackage
condition_then: click-free-giftPackage,condition-to-purchase,goto-home
condition_else: goto-home

[condition-in_location-freePackage]
extend: in_location
target: buttons\freePackage.png
args: 0.75

[click-free-giftPackage]
type: click
name: 每日礼包
target: buttons\freePackage.png
per: 0.75
sleep: 1
max_wait: 3

[condition-to-purchase]
type: condition
name: 条件判断是否有购买按钮
condition: condition-in_location-purchase
condition_then: click-purchase,click-get-item

[condition-in_location-purchase]
extend: in_location
target: buttons\purchase.png

[click-purchase]
type: click
name: 购买
target: buttons\purchase.png
sleep: 1
